**Goal:** trang `/view` (xem lưới ảnh, mở bài, đăng ảnh, nhãn trạng thái cụm) trong cùng trang web với trang giới thiệu, dựng tĩnh bằng Astro và Svelte, kiểm thử bằng trình duyệt tự động, và tự đăng lên `dr.qivarn.id.vn` qua kho `dr-preview` khi `main` qua hết kiểm tra.

**Architecture:** `app/web` là một dự án Astro xuất trang tĩnh. Trang giới thiệu giữ nguyên trong `public/` (chỉ thêm đoạn chuyển hướng và nút "Vào xem"). `/view` là một đảo Svelte: thư viện thuần (`src/lib`) lo trạng thái cụm, xếp lưới, nén ảnh, lưu trữ trên máy, kính lỏng; các component Svelte chỉ ghép chúng lại. Code gọi API sinh từ OpenAPI. `ci` dựng, kiểm thử bằng Playwright với API thật trong Docker, đo Lighthouse, rồi đẩy bản dựng sang `dr-preview`.

**Tech Stack:** Node 24, Astro 7.3.8, Svelte 5.57.2, @astrojs/svelte 9.0.1, @tanstack/svelte-query 6.3.1, openapi-typescript 7.13.0, openapi-fetch 0.17.0, TypeScript 5.9.3, Vitest 5.0.3, Playwright 1.64.0, @axe-core/playwright 4.13.0, @lhci/cli 0.15.1.

**Spec:** [`docs/thiet-ke-web.md`](../thiet-ke-web.md) mục 1, 3, 4, 5.

## Global Constraints

- Phông Be Vietnam Pro từ Google Fonts, nền đen, các biến màu và lớp `.glass` lấy đúng từ trang giới thiệu. Trang `/view` không tải Three.js.
- Đường dẫn: `/` trang giới thiệu, `/view/` trang dùng, một bài là `/view/?post={id}`, xem lại giới thiệu là `/?intro`.
- Khóa `localStorage`: `secret-code` (mã đăng bài), `entered-view` (`"1"` khi đã bấm Vào xem). Mọi đọc ghi bọc `try/catch`.
- Nhãn trạng thái: gọi `/healthz` mỗi 5 giây (2 giây khi sự cố), chờ tối đa 3 giây, dừng khi tab ẩn; hai lần lỗi liên tiếp mới chuyển sang sự cố.
- Ảnh gửi đi luôn được mã hóa lại trong trình duyệt: cạnh dài tối đa 2560 px, WebP chất lượng 0,85 giảm dần tới khi không quá 1 048 576 byte; không ghi được WebP thì dùng JPEG.
- `idempotency_key` tạo một lần cho mỗi lần bấm Đăng, giữ nguyên khi thử lại.
- Lighthouse `/view/` cấu hình điện thoại: hiệu năng ≥ 0,95. axe: không có lỗi mức `serious` hoặc `critical`.
- Biến dựng: `PUBLIC_API_URL`, `PUBLIC_API_READY` (`true` mới mở trang dùng).
- Không viết chú thích trong code (quy tắc của kho).

```ts
import createClient from 'openapi-fetch';

import type { components, paths } from './schema';

export type Post = components['schemas']['Post'];
export type PostPage = components['schemas']['PostPage'];
export type Health = components['schemas']['Health'];

export interface UploadResult {
  status: number;
  post?: Post;
  error?: { error: string; message: string };
}

export function createApiClient(base: string) {
  return createClient<paths>({ baseUrl: base });
}

export const imageUrl = (base: string, id: string): string => `${base}/v1/posts/${id}/image`;
export const thumbnailUrl = (base: string, id: string): string => `${base}/v1/posts/${id}/thumbnail`;

export async function verifyCode(base: string, code: string): Promise<number> {
  try {
    const r = await fetch(`${base}/v1/secret-code/verify`, { method: 'POST', headers: { Authorization: `Bearer ${code}` } });
    return r.status;
  } catch {
    return 0;
  }
}

export function uploadPost(
  base: string,
  code: string,
  data: { idempotencyKey: string; title: string; description: string; image: Blob },
  progress: (fraction: number) => void,
): Promise<UploadResult> {
  return new Promise((done) => {
    const form = new FormData();
    form.append('idempotency_key', data.idempotencyKey);
    form.append('title', data.title);
    form.append('description', data.description);
    form.append('image', data.image, data.image.type === 'image/jpeg' ? 'image.jpg' : 'image.webp');
    const x = new XMLHttpRequest();
    x.open('POST', `${base}/v1/posts`);
    x.setRequestHeader('Authorization', `Bearer ${code}`);
    x.timeout = 60_000;
    x.upload.onprogress = (e) => {
      if (e.lengthComputable) progress(e.loaded / e.total);
    };
    x.onload = () => {
      let body: unknown = null;
      try {
        body = JSON.parse(x.responseText);
      } catch {
        body = null;
      }
      if (x.status === 200 || x.status === 201) {
        done({ status: x.status, post: body as Post });
        return;
      }
      const error = (body as UploadResult['error']) ?? { error: 'unavailable', message: 'Không gửi được, thử lại sau.' };
      done({ status: x.status, error });
    };
    const onNetworkError = () => done({ status: 0, error: { error: 'network', message: 'Mất kết nối, thử lại.' } });
    x.onerror = onNetworkError;
    x.ontimeout = onNetworkError;
    x.send(form);
  });
}
```

```ts
export type Status =
  | { kind: 'checking' }
  | { kind: 'up'; cluster: string; region: string; readOnly: boolean }
  | { kind: 'outage'; since: number };

export type HealthEvent =
  | { kind: 'success'; cluster: string; region: string; readOnly: boolean; at: number }
  | { kind: 'failure'; at: number };

export interface Machine {
  status: Status;
  consecutiveFailures: number;
  firstFailureAt: number | null;
  notice: string | null;
}

export const newMachine = (): Machine => ({ status: { kind: 'checking' }, consecutiveFailures: 0, firstFailureAt: null, notice: null });

export function formatDuration(ms: number): string {
  const seconds = Math.max(0, Math.round(ms / 1000));
  const minutes = Math.floor(seconds / 60);
  return minutes > 0 ? `${minutes} phút ${seconds % 60} giây` : `${seconds} giây`;
}

export function step(m: Machine, s: HealthEvent): Machine {
  if (s.kind === 'failure') {
    const firstFailureAt = m.firstFailureAt ?? s.at;
    const consecutiveFailures = m.consecutiveFailures + 1;
    const status: Status = consecutiveFailures >= 2 ? { kind: 'outage', since: firstFailureAt } : m.status;
    return { status, consecutiveFailures, firstFailureAt, notice: m.notice };
  }
  const notice = m.status.kind === 'outage' ? `Đã chạy lại sau ${formatDuration(s.at - m.status.since)}` : m.notice;
  return { status: { kind: 'up', cluster: s.cluster, region: s.region, readOnly: s.readOnly }, consecutiveFailures: 0, firstFailureAt: null, notice };
}

export const dismissNotice = (m: Machine): Machine => ({ ...m, notice: null });

export const pollInterval = (m: Machine): number => (m.status.kind === 'outage' ? 2000 : 5000);

const REGION_NAMES: Record<string, string> = { koreacentral: 'Hàn Quốc', japanwest: 'Nhật Bản' };

export function badge(t: Status): { text: string; color: 'green' | 'amber' | 'grey' } {
  if (t.kind === 'checking') return { text: 'Đang kiểm tra…', color: 'grey' };
  if (t.kind === 'outage') return { text: 'Đang khôi phục…', color: 'amber' };
  const cluster = t.cluster === 'primary' ? 'Cụm chính' : 'Cụm dự phòng';
  const region = REGION_NAMES[t.region] ?? t.region;
  return { text: region ? `${cluster} · ${region}` : cluster, color: 'green' };
}
```

```ts
export interface KeyValueStore {
  getItem(k: string): string | null;
  setItem(k: string, v: string): void;
  removeItem(k: string): void;
}

const CODE_KEY = 'secret-code';
const ENTERED_VIEW_KEY = 'entered-view';

function openStore(): KeyValueStore | null {
  try {
    return window.localStorage;
  } catch {
    return null;
  }
}

function safely<T>(fn: () => T, fallback: T): T {
  try {
    return fn();
  } catch {
    return fallback;
  }
}

export const readCode = (store: KeyValueStore | null = openStore()): string | null => safely(() => store?.getItem(CODE_KEY) ?? null, null);
export const saveCode = (code: string, store: KeyValueStore | null = openStore()): void => safely(() => store?.setItem(CODE_KEY, code), undefined);
export const forgetCode = (store: KeyValueStore | null = openStore()): void => safely(() => store?.removeItem(CODE_KEY), undefined);
export const markEnteredView = (store: KeyValueStore | null = openStore()): void => safely(() => store?.setItem(ENTERED_VIEW_KEY, '1'), undefined);
```

```ts
export interface GridItem {
  id: string;
  width: number;
  height: number;
}

export interface Tile {
  id: string;
  x: number;
  y: number;
  width: number;
  height: number;
}

export function layoutGrid(posts: GridItem[], containerWidth: number, targetHeight: number, gap: number, caption: number): { tiles: Tile[]; totalHeight: number } {
  const tiles: Tile[] = [];
  let y = 0;
  let row: GridItem[] = [];
  let ratioSum = 0;
  const flushRow = (height: number) => {
    let x = 0;
    for (const a of row) {
      const width = (a.width / a.height) * height;
      tiles.push({ id: a.id, x, y, width, height });
      x += width + gap;
    }
    y += height + caption + gap;
    row = [];
    ratioSum = 0;
  };
  for (const a of posts) {
    row.push(a);
    ratioSum += a.width / a.height;
    if (ratioSum * targetHeight + gap * (row.length - 1) >= containerWidth) {
      flushRow((containerWidth - gap * (row.length - 1)) / ratioSum);
    }
  }
  if (row.length) flushRow(Math.min(targetHeight, (containerWidth - gap * (row.length - 1)) / ratioSum));
  return { tiles, totalHeight: Math.max(0, y - gap) };
}

export function targetHeight(containerWidth: number): number {
  return containerWidth < 600 ? Math.round((containerWidth - 8) / 2 / 1.33) : 240;
}
```

```ts
export const MAX_BYTES = 1_048_576;
export const MAX_EDGE = 2560;
const QUALITIES = [0.85, 0.75, 0.65, 0.55, 0.45];

export type Encoder = (width: number, height: number, quality: number) => Promise<Blob>;

export function scaledSize(width: number, height: number, edge = MAX_EDGE): [number, number] {
  const scale = Math.min(1, edge / Math.max(width, height));
  return [Math.max(1, Math.round(width * scale)), Math.max(1, Math.round(height * scale))];
}

export async function encodeWithinLimit(width: number, height: number, encode: Encoder, limit = MAX_BYTES): Promise<Blob> {
  let [r, c] = scaledSize(width, height);
  for (let attempt = 0; attempt < 6; attempt++) {
    for (const q of QUALITIES) {
      const b = await encode(r, c, q);
      if (b.size <= limit) return b;
    }
    [r, c] = [Math.max(1, Math.round(r * 0.8)), Math.max(1, Math.round(c * 0.8))];
  }
  throw new Error('Không nén được ảnh xuống dưới 1 MB.');
}

export async function compressImage(file: Blob): Promise<Blob> {
  const image = await createImageBitmap(file, { imageOrientation: 'from-image' });
  try {
    let type: 'image/webp' | 'image/jpeg' = 'image/webp';
    const encode: Encoder = async (r, c, q) => {
      const canvas = new OffscreenCanvas(r, c);
      const g = canvas.getContext('2d');
      if (!g) throw new Error('Trình duyệt không vẽ được ảnh.');
      g.drawImage(image, 0, 0, r, c);
      let b = await canvas.convertToBlob({ type: type, quality: q });
      if (b.type !== type) {
        type = 'image/jpeg';
        b = await canvas.convertToBlob({ type: type, quality: q });
      }
      return b;
    };
    return await encodeWithinLimit(image.width, image.height, encode);
  } finally {
    image.close();
  }
}
```

```ts
const NS = 'http://www.w3.org/2000/svg';

interface GlassPane {
  el: HTMLElement;
  bend: number;
  w: number;
  h: number;
  last: string;
  im?: SVGFEImageElement;
  dm?: SVGFEDisplacementMapElement;
  filter?: SVGFilterElement;
}

const panes = new Set<GlassPane>();
let defs: SVGSVGElement | null = null;
let counter = 0;
let px = -1;
let py = -1;
let running = false;

export function refractionAt(x: number, y: number, hx: number, hy: number, rr: number, band: number): [number, number] {
  const qx = Math.abs(x) - (hx - rr);
  const qy = Math.abs(y) - (hy - rr);
  let nx: number;
  let ny: number;
  let dist: number;
  if (qx > 0 && qy > 0) {
    const l = Math.hypot(qx, qy) || 1;
    dist = rr - l;
    nx = qx / l;
    ny = qy / l;
  } else if (qx > qy) {
    dist = rr - qx;
    nx = 1;
    ny = 0;
  } else {
    dist = rr - qy;
    nx = 0;
    ny = 1;
  }
  nx *= Math.sign(x) || 1;
  ny *= Math.sign(y) || 1;
  const t = Math.max(0, 1 - Math.max(0, dist) / band);
  const m = t * t * (3 - 2 * t);
  return [Math.round(128 - nx * m * 127), Math.round(128 - ny * m * 127)];
}

function refractionMap(w: number, h: number, r: number, band: number): string {
  const k = 0.5;
  const W = Math.max(2, Math.round(w * k));
  const H = Math.max(2, Math.round(h * k));
  const c = document.createElement('canvas');
  c.width = W;
  c.height = H;
  const g = c.getContext('2d');
  if (!g) return '';
  const img = g.createImageData(W, H);
  const d = img.data;
  const hx = w / 2;
  const hy = h / 2;
  const rr = Math.min(r, hx, hy);
  for (let y = 0; y < H; y++) {
    for (let x = 0; x < W; x++) {
      const [vr, vg] = refractionAt((x + 0.5) / k - hx, (y + 0.5) / k - hy, hx, hy, rr, band);
      const i = (y * W + x) * 4;
      d[i] = vr;
      d[i + 1] = vg;
      d[i + 2] = 128;
      d[i + 3] = 255;
    }
  }
  g.putImageData(img, 0, 0);
  return c.toDataURL();
}

function supportsRefraction(): boolean {
  const ud = (navigator as Navigator & { userAgentData?: { brands?: { brand: string }[] } }).userAgentData;
  const brands = ud?.brands?.map((b) => b.brand).join(' ') ?? '';
  return /Chromium|Google Chrome|Microsoft Edge/.test(brands);
}

function setAttrs(el: Element, attrs: Record<string, string>) {
  for (const [k, v] of Object.entries(attrs)) el.setAttribute(k, v);
}

function resize(p: GlassPane) {
  if (!p.im) return;
  const w = p.el.offsetWidth;
  const h = p.el.offsetHeight;
  if (!w || !h || (w === p.w && h === p.h)) return;
  p.w = w;
  p.h = h;
  const r = parseFloat(getComputedStyle(p.el).borderTopLeftRadius) || h / 2;
  setAttrs(p.im, { width: String(w), height: String(h), href: refractionMap(w, h, Math.min(r, h / 2), Math.min(24, h * 0.42)) });
}

function tick() {
  for (const p of panes) {
    const opacity = p.el.style.opacity === '' ? 1 : Number(p.el.style.opacity);
    if (p.dm) {
      const s = (p.bend * Math.min(1, opacity * 1.25)).toFixed(1);
      if (s !== p.last) {
        p.dm.setAttribute('scale', s);
        p.last = s;
      }
    }
    if (px >= 0 && opacity > 0.05) {
      const r = p.el.getBoundingClientRect();
      p.el.style.setProperty('--gx', `${Math.min(130, Math.max(-30, ((px - r.left) / r.width) * 100)).toFixed(1)}%`);
      p.el.style.setProperty('--gy', `${Math.min(170, Math.max(-70, ((py - r.top) / r.height) * 100)).toFixed(1)}%`);
    }
  }
  if (panes.size) requestAnimationFrame(tick);
  else running = false;
}

export function glass(el: HTMLElement, options: { bend?: number; blur?: number } = {}) {
  const p: GlassPane = { el, bend: options.bend ?? 28, w: 0, h: 0, last: '' };
  if (supportsRefraction()) {
    if (!defs) {
      defs = document.createElementNS(NS, 'svg');
      setAttrs(defs, { width: '0', height: '0', 'aria-hidden': 'true' });
      defs.style.position = 'absolute';
      document.body.prepend(defs);
      addEventListener('pointermove', (e) => {
        px = e.clientX;
        py = e.clientY;
      }, { passive: true });
    }
    const id = `glass${counter++}`;
    const filter = document.createElementNS(NS, 'filter');
    setAttrs(filter, { id, x: '0', y: '0', width: '100%', height: '100%', 'color-interpolation-filters': 'sRGB' });
    const im = document.createElementNS(NS, 'feImage');
    setAttrs(im, { x: '0', y: '0', preserveAspectRatio: 'none', result: 'map' });
    const dm = document.createElementNS(NS, 'feDisplacementMap');
    setAttrs(dm, { in: 'SourceGraphic', in2: 'map', xChannelSelector: 'R', yChannelSelector: 'G', scale: '0' });
    filter.append(im, dm);
    defs.append(filter);
    Object.assign(p, { im, dm, filter });
    el.style.backdropFilter = `url(#${id}) blur(${options.blur ?? 4}px) saturate(1.8)`;
  }
  panes.add(p);
  const ro = new ResizeObserver(() => resize(p));
  ro.observe(el);
  resize(p);
  if (!running) {
    running = true;
    requestAnimationFrame(tick);
  }
  return {
    destroy() {
      ro.disconnect();
      panes.delete(p);
      p.filter?.remove();
    },
  };
}
```

```ts
import { crc32, deflateSync } from 'node:zlib';

import type { APIRequestContext } from '@playwright/test';

export const API = process.env.PUBLIC_API_URL ?? 'http://127.0.0.1:8080';
export const CODE = process.env.TEST_SECRET_CODE ?? 'test-code-0123456789';

function chunk(kind: string, data: Buffer): Buffer {
  const length = Buffer.alloc(4);
  length.writeUInt32BE(data.length);
  const name = Buffer.from(kind, 'ascii');
  const crc = Buffer.alloc(4);
  crc.writeUInt32BE(crc32(Buffer.concat([name, data])) >>> 0);
  return Buffer.concat([length, name, data, crc]);
}

export function createPng(width: number, height: number): Buffer {
  const ihdr = Buffer.alloc(13);
  ihdr.writeUInt32BE(width, 0);
  ihdr.writeUInt32BE(height, 4);
  ihdr[8] = 8;
  ihdr[9] = 2;
  const raw = Buffer.alloc(height * (1 + width * 3));
  for (let y = 0; y < height; y++) {
    const offset = y * (1 + width * 3);
    for (let x = 0; x < width; x++) {
      const i = offset + 1 + x * 3;
      raw[i] = x % 256;
      raw[i + 1] = y % 256;
      raw[i + 2] = (x + y) % 256;
    }
  }
  return Buffer.concat([
    Buffer.from([0x89, 0x50, 0x4e, 0x47, 0x0d, 0x0a, 0x1a, 0x0a]),
    chunk('IHDR', ihdr),
    chunk('IDAT', deflateSync(raw)),
    chunk('IEND', Buffer.alloc(0)),
  ]);
}

export async function postViaApi(request: APIRequestContext, title: string, width: number, height: number): Promise<{ id: string; title: string }> {
  const r = await request.post(`${API}/v1/posts`, {
    headers: { Authorization: `Bearer ${CODE}` },
    multipart: {
      idempotency_key: crypto.randomUUID(),
      title: title,
      description: 'Bài thử tự động',
      image: { name: 'image.png', mimeType: 'image/png', buffer: createPng(width, height) },
    },
  });
  if (r.status() !== 201) throw new Error(`đăng thử lỗi ${r.status()}: ${await r.text()}`);
  return r.json();
}
```

```svelte
<script lang="ts">
  import { createInfiniteQuery, type InfiniteData, useQueryClient } from '@tanstack/svelte-query';

  import { type Post, createApiClient, type PostPage } from '../lib/api/client';
  import { dismissNotice, step, pollInterval, newMachine, type HealthEvent } from '../lib/status';
  import PhotoGrid from './PhotoGrid.svelte';
  import TopBar from './TopBar.svelte';
  import Toast from './Toast.svelte';
  import PostViewer from './PostViewer.svelte';

  let { apiUrl }: { apiUrl: string } = $props();
  const api = createApiClient(apiUrl);
  const queryClient = useQueryClient();
  let machine = $state(newMachine());
  let uploadOpen = $state(false);
  let openPostId = $state<string | null>(new URLSearchParams(location.search).get('post'));

  const posts = createInfiniteQuery(() => ({
    queryKey: ['post'],
    initialPageParam: null as string | null,
    queryFn: async ({ pageParam }): Promise<PostPage> => {
      const query = pageParam ? { limit: 24, before: pageParam } : { limit: 24 };
      const { data, error } = await api.GET('/v1/posts', { params: { query } });
      if (error || !data) throw new Error(error?.message ?? 'Không tải được danh sách.');
      return data;
    },
    getNextPageParam: (page: PostPage) => page.next ?? undefined,
  }));
  const allPosts = $derived(posts.data?.pages.flatMap((p) => p.posts) ?? []);

  async function fetchPost(id: string): Promise<Post | null> {
    const { data } = await api.GET('/v1/posts/{id}', { params: { path: { id } } });
    return data ?? null;
  }

  $effect(() => {
    let cancelled = false;
    let timer: ReturnType<typeof setTimeout> | undefined;
    async function check() {
      if (cancelled) return;
      if (document.hidden) {
        timer = setTimeout(check, 1000);
        return;
      }
      let event: HealthEvent;
      try {
        const r = await fetch(`${apiUrl}/healthz`, { signal: AbortSignal.timeout(3000), cache: 'no-store' });
        const v = await r.json();
        event = r.ok
          ? { kind: 'success', cluster: v.cluster, region: v.region, readOnly: v.read_only, at: Date.now() }
          : { kind: 'failure', at: Date.now() };
      } catch {
        event = { kind: 'failure', at: Date.now() };
      }
      if (cancelled) return;
      const previousKind = machine.status.kind;
      machine = step(machine, event);
      if (previousKind === 'outage' && machine.status.kind === 'up') queryClient.invalidateQueries({ queryKey: ['post'] });
      timer = setTimeout(check, pollInterval(machine));
    }
    check();
    return () => {
      cancelled = true;
      clearTimeout(timer);
    };
  });

  $effect(() => {
    if (!machine.notice) return;
    const h = setTimeout(() => (machine = dismissNotice(machine)), 10_000);
    return () => clearTimeout(h);
  });

  $effect(() => {
    const syncFromUrl = () => (openPostId = new URLSearchParams(location.search).get('post'));
    addEventListener('popstate', syncFromUrl);
    return () => removeEventListener('popstate', syncFromUrl);
  });

  function openPost(id: string) {
    history.pushState({ post: id }, '', `?post=${id}`);
    openPostId = id;
  }

  function switchPost(id: string) {
    history.replaceState(history.state, '', `?post=${id}`);
    openPostId = id;
  }

  function close() {
    if (!openPostId) return;
    openPostId = null;
    if (history.state?.post) history.back();
    else history.replaceState(null, '', location.pathname);
  }

  function prepend(b: Post) {
    queryClient.setQueryData<InfiniteData<PostPage>>(['post'], (old) =>
      old
        ? { ...old, pages: [{ ...old.pages[0], posts: [b, ...old.pages[0].posts.filter((x) => x.id !== b.id)] }, ...old.pages.slice(1)] }
        : old,
    );
  }
</script>

<TopBar status={machine.status} onUpload={() => (uploadOpen = true)} />

<main>
  <h1>Ảnh của nhóm</h1>
  <p class="subtitle">Mỗi ảnh được cất ở kho chính và chép sang Nhật Bản.</p>
  {#if posts.isPending && machine.status.kind !== 'outage'}
    <p class="empty-state" role="status">Đang tải ảnh…</p>
  {:else if posts.isError && allPosts.length === 0}
    <p class="empty-state" role="status">Chưa tải được ảnh. Trang sẽ tự tải lại khi máy chủ chạy lại.</p>
  {:else if posts.isSuccess && allPosts.length === 0}
    <p class="empty-state">Chưa có ảnh nào.</p>
  {/if}
  <PhotoGrid
    posts={allPosts}
    {apiUrl}
    hasMore={posts.hasNextPage}
    loadingMore={posts.isFetchingNextPage}
    loadMore={() => posts.fetchNextPage()}
    onOpen={openPost}
  />
</main>

{#if openPostId}
  <PostViewer posts={allPosts} id={openPostId} {apiUrl} {fetchPost} onClose={close} onSwitch={switchPost} />
{/if}

{#if uploadOpen}
  <Toast text="Đăng ảnh sắp có." kind="done" onClose={() => (uploadOpen = false)} />
{/if}

{#if machine.status.kind === 'outage'}
  <Toast text="Máy chủ đang được khôi phục. Ảnh đã tải vẫn xem được; trang tự thử lại." kind="outage" />
{:else if machine.notice}
  <Toast text={machine.notice} kind="done" onClose={() => (machine = dismissNotice(machine))} />
{/if}
```

```svelte
<script lang="ts">
  import { type Post, uploadPost, verifyCode } from '../lib/api/client';
  import { glass } from '../lib/glass';
  import { readCode, saveCode, forgetCode } from '../lib/storage';
  import { compressImage } from '../lib/compress-image';

  let { apiUrl, onDone, onClose }: { apiUrl: string; onDone: (b: Post) => void; onClose: () => void } = $props();

  let dialog: HTMLDialogElement | undefined = $state();
  let code = $state<string | null>(readCode());
  let codeInput = $state('');
  let file = $state<File | null>(null);
  let previewUrl = $state<string | null>(null);
  let title = $state('');
  let description = $state('');
  let progress = $state<number | null>(null);
  let error = $state<string | null>(null);
  let sending = $state(false);
  let idempotencyKey: string | null = null;

  $effect(() => {
    if (dialog && !dialog.open) dialog.showModal();
  });

  $effect(() => {
    const url = previewUrl;
    return () => {
      if (url) URL.revokeObjectURL(url);
    };
  });

  async function submitCode(e: SubmitEvent) {
    e.preventDefault();
    error = null;
    const entered = codeInput.trim();
    const s = await verifyCode(apiUrl, entered);
    if (s === 204) {
      saveCode(entered);
      code = entered;
      codeInput = '';
    } else if (s === 429) {
      error = 'Thử sai quá nhiều lần. Đợi một lúc rồi thử lại.';
    } else if (s === 401) {
      error = 'Mã không đúng.';
    } else {
      error = 'Không kiểm tra được mã, thử lại sau.';
    }
  }

  function pick(f: File | null | undefined) {
    if (!f) return;
    if (!f.type.startsWith('image/')) {
      error = 'Hãy chọn một tệp ảnh.';
      return;
    }
    error = null;
    file = f;
    idempotencyKey = null;
    previewUrl = URL.createObjectURL(f);
  }

  function dropCode() {
    forgetCode();
    code = null;
  }

  async function submit(e: SubmitEvent) {
    e.preventDefault();
    if (!file || !code) return;
    error = null;
    sending = true;
    progress = 0;
    idempotencyKey ??= crypto.randomUUID();
    try {
      const image = await compressImage(file);
      const result = await uploadPost(apiUrl, code, { idempotencyKey, title, description, image }, (p) => (progress = p));
      if (result.post) {
        idempotencyKey = null;
        onDone(result.post);
        dialog?.close();
        return;
      }
      if (result.status === 401) {
        dropCode();
        error = 'Mã đã bị đổi, nhập lại mã.';
      } else {
        error = result.error?.message ?? 'Không đăng được, thử lại.';
      }
    } catch (err) {
      error = err instanceof Error ? err.message : 'Không xử lý được ảnh.';
    } finally {
      sending = false;
      progress = null;
    }
  }
</script>
```
